# Migration Finalization Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Finish the remaining `docs/migration.md` final acceptance gaps in one coherent pass: remove legacy transition code, add missing final-check scripts, expand E2E/coverage gates, and bring migration docs to an auditable completed state.

**Architecture:** Treat `requireServerSession` / `requireApiSession` as the only identity boundary and keep service-layer inputs as trusted route-composed payloads. Add executable finalization checks so documentation and scripts cannot drift back into a partially migrated state. Keep production/staging-only verification as explicit operation records under `docs/operations.md` and audit artifacts under `docs/audit/`.

**Tech Stack:** Next.js App Router route handlers, Node.js built-in test runner with `--experimental-strip-types`, Playwright, ESLint, PostgreSQL via `pg`, repository-local scripts under `scripts/`.

- [ ] **Step 3: Create dashboard usage audit script**

Create `scripts/audit-dashboard-usage.mjs`:

```js
import { mkdir, writeFile } from "node:fs/promises";
import path from "node:path";
import pg from "pg";

const { Client } = pg;
const root = process.cwd();
const outputPath = path.join(root, "docs/audit/dashboard-usage.md");
const databaseUrl = process.env.SDS_DATABASE_URL ?? process.env.DATABASE_URL;

const limits = {
  maxViews: Number.parseInt(process.env.SDS_QUOTA_VIEWS_PER_DASHBOARD ?? "50", 10),
  maxQueries: Number.parseInt(process.env.SDS_QUOTA_QUERIES_PER_DASHBOARD ?? "100", 10),
  maxDocumentBytes:
    Number.parseInt(process.env.SDS_QUOTA_DOCUMENT_SIZE_MB ?? "2", 10) * 1024 * 1024,
};

function tableRows(rows) {
  if (rows.length === 0) {
    return "| scope | dashboard_count | max_views | max_queries | max_doc_bytes | status |\n|---|---:|---:|---:|---:|---|\n| none | 0 | 0 | 0 | 0 | OK |";
  }

  return [
    "| scope | dashboard_count | max_views | max_queries | max_doc_bytes | status |",
    "|---|---:|---:|---:|---:|---|",
    ...rows.map((row) => {
      const maxViews = Number(row.max_views ?? 0);
      const maxQueries = Number(row.max_queries ?? 0);
      const maxBytes = Number(row.max_doc_bytes ?? 0);
      const status =
        maxViews > limits.maxViews ||
        maxQueries > limits.maxQueries ||
        maxBytes > limits.maxDocumentBytes
          ? "OVER_LIMIT"
          : "OK";
      return `| ${row.scope} | ${row.dashboard_count} | ${maxViews} | ${maxQueries} | ${maxBytes} | ${status} |`;
    }),
  ].join("\n");
}

async function main() {
  await mkdir(path.dirname(outputPath), { recursive: true });

  if (!databaseUrl) {
    await writeFile(
      outputPath,
      [
        "# Dashboard Usage Audit",
        "",
        "Status: NOT_RUN",
        "",
        "No `SDS_DATABASE_URL` or `DATABASE_URL` was configured when this audit ran.",
        "Run `SDS_DATABASE_URL=postgresql://... npm run audit:dashboard-usage` against staging before final release.",
        "",
      ].join("\n"),
      "utf8",
    );
    throw new Error("DATABASE_URL_REQUIRED_FOR_DASHBOARD_USAGE_AUDIT");
  }

  const client = new Client({ connectionString: databaseUrl });
  await client.connect();
  try {
    const result = await client.query(`
      with documents as (
        select 'drafts' as scope, dashboard_document
        from workspace_dashboard_drafts
        union all
        select 'published' as scope, dashboard_document
        from workspace_dashboard_published
      )
      select
        scope,
        count(*)::int as dashboard_count,
        coalesce(max(jsonb_array_length(dashboard_document->'dashboard_spec'->'views')), 0)::int as max_views,
        coalesce(max(jsonb_array_length(dashboard_document->'query_defs')), 0)::int as max_queries,
        coalesce(max(octet_length(dashboard_document::text)), 0)::int as max_doc_bytes
      from documents
      group by scope
      order by scope asc
    `);

    const body = [
      "# Dashboard Usage Audit",
      "",
      `Status: ${result.rows.some((row) => {
        const views = Number(row.max_views ?? 0);
        const queries = Number(row.max_queries ?? 0);
        const bytes = Number(row.max_doc_bytes ?? 0);
        return views > limits.maxViews || queries > limits.maxQueries || bytes > limits.maxDocumentBytes;
      }) ? "OVER_LIMIT" : "OK"}`,
      "",
      `Limits: views=${limits.maxViews}, queries=${limits.maxQueries}, document_bytes=${limits.maxDocumentBytes}`,
      "",
      tableRows(result.rows),
      "",
    ].join("\n");

    await writeFile(outputPath, body, "utf8");

    if (body.includes("OVER_LIMIT")) {
      throw new Error("DASHBOARD_USAGE_OVER_DEFAULT_QUOTA");
    }
  } finally {
    await client.end();
  }
}

main().catch((error) => {
  console.error(error instanceof Error ? error.message : String(error));
  process.exit(1);
});
```

- [ ] **Step 4: Create final acceptance script**

Create `scripts/check-final-acceptance.mjs`:

```js
import { existsSync, readFileSync } from "node:fs";
import path from "node:path";

const root = process.cwd();
const errors = [];

function read(relativePath) {
  return readFileSync(path.join(root, relativePath), "utf8");
}

function assertFileMissing(relativePath) {
  if (existsSync(path.join(root, relativePath))) {
    errors.push(`${relativePath} must not exist`);
  }
}

function assertFileExists(relativePath) {
  if (!existsSync(path.join(root, relativePath))) {
    errors.push(`${relativePath} must exist`);
  }
}

function assertNoMatch(relativePath, pattern, label) {
  const source = read(relativePath);
  if (pattern.test(source)) {
    errors.push(`${relativePath} contains ${label}`);
  }
}

assertFileMissing("src/server/request-context.ts");
assertFileMissing("src/web/auth/auth-session.ts");
assertFileExists("docs/archive/migration-2026-q2.md");
assertFileExists("docs/operations.md");
assertFileExists("docs/audit/dashboard-usage.md");
assertNoMatch("docs/architecture.md", /🟡|🔴/, "non-final status markers");
assertNoMatch("docs/audit/route-inventory.md", /\[ \]/, "unchecked route inventory rows");

for (const file of [
  "src/server/dashboards/service.ts",
  "src/server/authoring/editing-session-service.ts",
  "src/server/authoring/chat-request.ts",
]) {
  assertNoMatch(file, /resolveServerRequestContext|@\/server\/request-context/, "legacy request context");
}

const packageJson = JSON.parse(read("package.json"));
for (const scriptName of [
  "script:check-i18n",
  "audit:dashboard-usage",
  "check:final",
]) {
  if (!packageJson.scripts?.[scriptName]) {
    errors.push(`package.json missing ${scriptName}`);
  }
}

if (packageJson.scripts?.["test:contract"] !== "node scripts/run-contract-tests.mjs") {
  errors.push("package.json test:contract must use scripts/run-contract-tests.mjs");
}

if (errors.length > 0) {
  console.error(errors.join("\n"));
  process.exit(1);
}

console.log("Final acceptance static check passed.");
```

```ts
import { expect, test } from "@playwright/test";

test("unauthenticated protected APIs reject requests", async ({ request }) => {
  const response = await request.get("/api/dashboards");
  expect(response.status()).toBe(401);
  const payload = await response.json();
  expect(payload.reason).toBe("AUTH_REQUIRED");
});

test("login session read refresh and logout use http-only cookie flow", async ({ request, baseURL }) => {
  const origin = baseURL ?? "http://127.0.0.1:3000";
  const login = await request.post("/api/auth/login", {
    headers: {
      origin,
      "content-type": "application/json",
    },
    data: {
      method: "account",
      identity: "alice@example.com",
      password: "local-dev",
    },
  });
  expect(login.status()).toBe(200);

  const session = await request.get("/api/auth/session");
  expect(session.status()).toBe(200);
  const sessionPayload = await session.json();
  expect(sessionPayload.data.workspace_id).toBe("ws_default");
  expect(sessionPayload.data.permissions).toContain("dashboard.read");

  const refresh = await request.post("/api/auth/refresh", {
    headers: { origin },
  });
  expect(refresh.status()).toBe(200);

  const logout = await request.post("/api/auth/logout", {
    headers: { origin },
  });
  expect(logout.status()).toBe(200);

  const afterLogout = await request.get("/api/auth/session");
  expect(afterLogout.status()).toBe(401);
});

test("cross-site mutating request is denied by CSRF origin policy", async ({ request }) => {
  const response = await request.post("/api/auth/login", {
    headers: {
      origin: "https://attacker.example",
      "content-type": "application/json",
    },
    data: {
      method: "account",
      identity: "alice@example.com",
      password: "local-dev",
    },
  });

  expect(response.status()).toBe(403);
  const payload = await response.json();
  expect(payload.reason).toBe("CSRF_ORIGIN_DENIED");
});
```
